<script>
https://script.google.com/macros/s/AKfycbyeihm_x75VXP_rBRexummbt6hvpp3g6JEiVnlAA_fIyuLwX0wUnWGCOJCCPxKJ0ifL/exec 
const GAS_WEB_APP_URL = "YOUR_APPS_SCRIPT_WEB_APP_URL_HERE"; 

let currentTeacherUser = "";

function openTeacherModal() {
  document.getElementById("teacherModal").classList.remove("hidden");
}

function closeTeacherModal() {
  document.getElementById("teacherModal").classList.add("hidden");
}

async function submitTeacherLogin() {
  const username = document.getElementById("teacherUsername").value.trim();
  const password = document.getElementById("teacherPassword").value.trim();

  if (!username || !password) {
    alert("กรุณากรอกชื่อผู้ใช้และรหัสผ่าน");
    return;
  }

  const response = await fetch(GAS_WEB_APP_URL, {
    method: "POST",
    body: JSON.stringify({
      action: "teacherLogin",
      username: username,
      password: password
    })
  });

  const result = await response.json();
  if (result.status === "success") {
    currentTeacherUser = result.username;
    document.getElementById("displayName").innerText = result.name;
    document.getElementById("loginSection").classList.add("hidden");
    document.getElementById("changePasswordSection").classList.remove("hidden");
  } else {
    alert(result.message);
  }
}

async function submitChangePassword() {
  const oldPassword = document.getElementById("oldPassword").value.trim();
  const newPassword = document.getElementById("newPassword").value.trim();

  if (!oldPassword || !newPassword) {
    alert("กรุณากรอกข้อมูลให้ครบถ้วน");
    return;
  }

  const response = await fetch(GAS_WEB_APP_URL, {
    method: "POST",
    body: JSON.stringify({
      action: "changePassword",
      username: currentTeacherUser,
      oldPassword: oldPassword,
      newPassword: newPassword
    })
  });

  const result = await response.json();
  alert(result.message);
  
  if (result.status === "success") {
    document.getElementById("oldPassword").value = "";
    document.getElementById("newPassword").value = "";
    logoutTeacher();
  }
}

function logoutTeacher() {
  currentTeacherUser = "";
  document.getElementById("teacherUsername").value = "";
  document.getElementById("teacherPassword").value = "";
  document.getElementById("changePasswordSection").classList.add("hidden");
  document.getElementById("loginSection").classList.remove("hidden");
  closeTeacherModal();
}
</script>
